\section{Proof of Lemma~\ref{M-lem:prime-sum}, and the kernel bound}\label{app:prime-sum}

\begin{proof}
Throughout, $Q \ge 4$, so that $\log Q > 1$, and $L_\chi$ stands for $L(\cdot, \chi)$, which is entire since $\chi$ is
not trivial. Put $\sigma_* = 11/12 + \delta$ and $\tau = (1 + H)^{-J}$.

\emph{The weight.} Let $f_Q(u) = f_0(u) \log Q/\log(uQ)$ where $f_0(u) \ne 0$, which needs $u \ge 1/2$ and so
$uQ \ge 2$, and $f_Q(u) = 0$ elsewhere; and let $w_y(u) = u^{-iy} f_Q(u)$. If
$f_0(n/Q) \ne 0$ then $n \ge Q/2 \ge 2$ and $f_Q(n/Q) = f_0(n/Q) \log Q/\log n$, so that
$\Lambda(n) f_Q(n/Q) = \log Q \cdot (\Lambda(n)/\log n) f_0(n/Q)$. Since $\Lambda(p)/\log p = 1$,
\begin{gather*}
  \sum_n \chi(n) \Lambda(n) w_y(n/Q) = \log Q \, \bigl(S_Q(\chi, y) + R\bigr), \\
  R = \sum_{\substack{p^k \le 2Q \\ k \ge 2}} \frac{\chi(p^k)}{k} \Bigl(\frac{p^k}{Q}\Bigr)^{-iy} f_0\Bigl(\frac{p^k}{Q}\Bigr).
\end{gather*}
Each term of $R$ is at most $C = \max |f_0|$ in absolute value, and a proper prime power $p^k \le 2Q$ has
$p \le \sqrt{2Q}$ and $k \le \log_2(2Q)$, so $|R| \le C(\sqrt{2Q} + 1)(\log_2(2Q) + 1)$. It remains to prove
\begin{equation}\label{eq:Sw-bound}
  \Bigl|\sum_n \chi(n) \Lambda(n) w_y(n/Q)\Bigr| \le M_\delta(\theta) \, \ell_{q, y} \, B \bigl(Q^{\theta + \delta} + Q^{\sigma_*} \tau\bigr).
\end{equation}

\emph{The Mellin transform.} Write $\mathcal M f(s) = \int_0^\infty u^{s - 1} f(u) \, du$. The substitution
$u = e^{-v}$ gives, with $c_Q = 1/\log Q \in (0, 1/\log 4]$,
\[
  \mathcal M f_Q(\sigma + it) = \int_{\mathbb R} e^{-itv} V(v) \, dv, \qquad
  V(v) = e^{-\sigma v} f_0(e^{-v}) (1 - c_Q v)^{-1}.
\]
The function $V$ vanishes for $|v| > \log 2$, and $c_Q v \le 1/2$ for $v \le \log 2$; the pole $v = \log Q \ge \log 4$ of
$(1 - c_Q v)^{-1}$ lies where $V$ vanishes, so $V$ is smooth. By Leibniz's
rule, with $|(d/dv)^i (1 - c_Q v)^{-1}| = i! \, c_Q^i |1 - c_Q v|^{-1 - i} \le i! \, 2^{i + 1}$ on $|v| \le \log 2$,
the $j$-th derivative of $V$ is bounded by a constant $V_j$ that depends only on $f_0$ and $j$, uniformly in
$0 \le \sigma \le 2$ and $Q \ge 4$. Integration by parts $j$ times gives
$|t|^j |\mathcal M f_Q(\sigma + it)| \le 2 \log 2 \cdot V_j$, and with $j = 0$ and $j = J + 3$,
\[
  (1 + |t|)^{J + 3} |\mathcal M f_Q(\sigma + it)| \le B \qquad (0 \le \sigma \le 2,\ t \in \mathbb R,\ Q \ge 4),
\]
where $B = 2^{J + 4} \log 2 \, (V_0 + V_{J + 3})$ depends only on $f_0$ and $J$. Since $f_Q$ is continuous on
$(0, \infty)$ and vanishes near $0$ and near $\infty$, $\mathcal M f_Q$ is entire. Finally
$\mathcal M w_y(s) = \mathcal M f_Q(s - iy)$, so that
\begin{equation}\label{eq:mellin-decay}
  (1 + |t - y|)^{J + 3} |\mathcal M w_y(\sigma + it)| \le B \qquad (0 \le \sigma \le 2,\ t \in \mathbb R).
\end{equation}

\emph{Perron's formula.} For $\operatorname{Re} s > 1$, $-L_\chi'/L_\chi(s) = \sum_n \chi(n) \Lambda(n) n^{-s}$.
Mellin inversion on the line $\operatorname{Re} s = 2$ at each point $n/Q$, with the exchange of sum and integral
justified by \eqref{eq:mellin-decay}, gives
\begin{gather*}
  \sum_n \chi(n) \Lambda(n) w_y(n/Q) = \frac1{2\pi} \int_{\mathbb R} F(2 + it) \, dt, \\
  F(s) = -\frac{L_\chi'}{L_\chi}(s) \, \mathcal M w_y(s) \, Q^s.
\end{gather*}

\emph{The logarithmic derivative.} For $1/2 \le \theta_0 < \theta' < 3/2$ there is $C_{\log}(\theta_0, \theta')$ with
the following property, for every $q$, every non-trivial $\chi$ mod $q$ and every $t \in \mathbb R$: if $L_\chi$ has no
zero in the open disc $D_t$ of centre $3/2 + it$ and radius $\rho = 3/2 - \theta_0$, then
\begin{equation}\label{eq:logderiv}
  \Bigl|\frac{L_\chi'}{L_\chi}(\sigma + it)\Bigr| \le C_{\log}(\theta_0, \theta') \bigl(\log(q + 2) + \log(|t| +
  4)\bigr) \qquad (\sigma \ge \theta').
\end{equation}
For $\sigma \ge 3/2$ this holds as soon as $C_{\log}(\theta_0, \theta') \ge -\zeta'/\zeta(3/2)$, by the Dirichlet series
of $L_\chi'/L_\chi$. For $\theta' \le \sigma < 3/2$ it follows from a lemma of Landau's type: if $f$ is analytic near
the closed unit disc, $f(0) = 1$, $|f| \le B_0$ on $|z| \le R$ with $B_0 > 1$, $f$ has no zero in $|z| < 1$ and
$0 < r' < r < R' < R < 1$, then
\[
  |f'/f(z)| \le \Bigl(\frac{16 r^2}{(r - r')^3} + \frac1{(R^2/R' - R') \log(R/R')}\Bigr) \log B_0 \qquad (|z| \le r').
\]
(The library uses the form of this lemma in PrimeNumberTheoremAnd \cite{pnt-plus}, \lean{FinalBound}, which combines a
Blaschke product, the Borel--Carath\'eodory inequality and Jensen's formula, and allows zeros.) Apply it to
$f(z) = L_\chi(3/2 + it + \rho z)/L_\chi(3/2 + it)$ at $z = (\sigma - 3/2)/\rho$, with $r' = (3/2 - \theta')/\rho$ and
with $r$, $R'$ and $R$ a quarter, a half and three quarters of the way from $r'$ to $1$; then
$f'/f(z) = \rho \, L_\chi'/L_\chi(\sigma + it)$. The bound $B_0$ comes from two estimates. First,
$|L_\chi(s)| \le 3(q + 1)(|\operatorname{Im} s| + 2)$ for $1/2 \le \operatorname{Re} s \le 3$, which follows from the
partial-summation bound $|L_\chi(s)| \le (q + 1)|s|/\operatorname{Re} s$ for $\operatorname{Re} s > 0$. Second,
$|L_\chi(3/2 + it)|^{-1} \le \zeta(3/2)$, from $1/L_\chi(s) = \sum \mu(n) \chi(n) n^{-s}$. Together they give
$|f| \le B_0 = 2 + 6(q + 1)\zeta(3/2)(|t| + 4)$ on $|z| \le R$, and $\log B_0 \ll \log(q + 2) + \log(|t| + 4)$. The
constant $C_{\log}(\theta_0, \theta')$ is explicit and of order $(\theta' - \theta_0)^{-3}$.

The disc $D_t$ lies in $\operatorname{Re} s > \theta_0$ and, since $\rho \le 1$, in $|\operatorname{Im} s - t| < 1$. So
\eqref{eq:logderiv} applies in two cases:
\begin{enumerate}
\item[(i)] with $\theta_0 = 11/12$ and every $t$, by Input~\ref{M-inp:half-plane};
\item[(ii)] with $\theta_0 = \theta$ and every $t$ with $|t - y| \le H$, by the hypothesis of the lemma.
\end{enumerate}
Since $|t| + 4 \le (|y| + 4)(1 + |t - y|)$ and $\log(1 + u) \le u$, we have
$\log(q + 2) + \log(|t| + 4) \le \ell_{q, y}(1 + |t - y|)$. With \eqref{eq:mellin-decay} this gives, in either case,
\begin{equation}\label{eq:F-bound}
  |F(\sigma + it)| \le C_{\log}(\theta_0, \theta') \, \ell_{q, y} \, B \, Q^\sigma (1 + |t - y|)^{-J - 2} \qquad (\theta'
  \le \sigma \le 2).
\end{equation}

\emph{The contour.} By Input~\ref{M-inp:half-plane}, $F$ is holomorphic on $\sigma_* \le \operatorname{Re} s \le 2$, and by
\eqref{eq:F-bound} in case (i), with $\theta' = \sigma_*$, it is $O((1 + |t - y|)^{-2})$ there uniformly. So Cauchy's
theorem on the rectangles $[\sigma_*, 2] \times [-t_0, t_0]$, with $t_0 \to \infty$, moves the line of integration from $2$ to
$\sigma_*$. Split the line $\operatorname{Re} s = \sigma_*$ at the heights $y \pm H$. By the hypothesis of the lemma, $F$
is holomorphic on the rectangle $\theta + \delta \le \operatorname{Re} s \le \sigma_*$, $|\operatorname{Im} s - y| \le H$,
so by Cauchy's theorem the integral over $|t - y| \le H$ on the line $\sigma_*$ equals the integral over the same
heights on the line $\theta + \delta$ plus the integrals over the two horizontal sides at the heights $y \pm H$.

\emph{The pieces.} Put $C_1 = C_{\log}(11/12, \sigma_*)$ and $C_2 = C_{\log}(\theta, \theta + \delta)$.
\begin{itemize}
\item On the line $\sigma_*$ with $|t - y| \ge H$: by \eqref{eq:F-bound} in case (i) and
  $(1 + u)^{-J - 2} \le (1 + H)^{-J} (1 + u^2)^{-1}$ for $u \ge H$,
  $|F| \le C_1 \ell_{q, y} B Q^{\sigma_*} \tau/(1 + (t - y)^2)$. Each of the two tails contributes at most
  $\pi C_1 \ell_{q, y} B Q^{\sigma_*} \tau$.
\item On the line $\theta + \delta$ with $|t - y| \le H$: by \eqref{eq:F-bound} in case (ii),
  $|F| \le C_2 \ell_{q, y} B Q^{\theta + \delta}/(1 + (t - y)^2)$, so this part contributes at most
  $\pi C_2 \ell_{q, y} B Q^{\theta + \delta}$.
\item On each horizontal side, of length $11/12 - \theta \le 1$: by \eqref{eq:F-bound} in case (ii), with
  $Q^\sigma \le Q^{\sigma_*}$, $|F| \le C_2 \ell_{q, y} B Q^{\sigma_*} (1 + H)^{-J - 2} \le C_2 \ell_{q, y} B Q^{\sigma_*} \tau$.
\end{itemize}
Adding the pieces and dividing by $2\pi$, the sum in \eqref{eq:Sw-bound} is at most
\begin{multline*}
  C_1 \ell_{q, y} B Q^{\sigma_*} \tau + \tfrac12 C_2 \ell_{q, y} B Q^{\theta + \delta} + \tfrac1\pi C_2 \ell_{q, y} B
  Q^{\sigma_*} \tau \\
  \le (C_1 + C_2) \, \ell_{q, y} \, B \bigl(Q^{\theta + \delta} + Q^{\sigma_*} \tau\bigr),
\end{multline*}
which is \eqref{eq:Sw-bound} with $M_\delta(\theta) = C_{\log}(\theta, \theta + \delta) + C_{\log}(11/12, 11/12 +
\delta)$.
\end{proof}

\subsection*{The kernel bound of Section~\ref{M-sec:two-bounds}}

Let $\delta' = \varepsilon' = 1/1000$, let $\psi$ be a non-trivial character mod $q$, and let $Y \ge 1$. By Mellin
inversion, $k_Y(\psi) = (2\pi i)^{-1} \int_{(2)} L(s, \psi) \Gamma(s) Y^s \, ds$. Since $L(\cdot, \psi)$ is entire and
$1/12 - \delta' > 0$, the line moves to $\operatorname{Re} s = 1/12 - \delta'$ without crossing a pole. There the
functional equation of the primitive character $\psi^*$ that induces $\psi$, of conductor $q^*$, gives
$|L(s, \psi^*)| \ll (q^*(|\operatorname{Im} s| + 2))^{5/12 + \delta'} |L(1 - s, \overline{\psi^*})|$, and $1 - s$ lies
on $\operatorname{Re} s = 11/12 + \delta'$. There Input~\ref{M-inp:half-plane} gives the Lindelöf-type bound
$|L(s, \chi)|^{\pm 1} \ll ((q + 2)(|\operatorname{Im} s| + 4))^{\varepsilon'}$, uniformly in $q$ and $\chi \ne \chi_0$,
by the argument by which the Riemann hypothesis implies the Lindelöf hypothesis \cite[\S 14.2]{titchmarsh}. The lemma
of Landau's type above bounds $|L'/L(s, \chi)|$ by a multiple of $\log((q + 2)(|\operatorname{Im} s| + 4))$ on
$\operatorname{Re} s \ge 11/12 + \delta'/2$; the three-lines theorem, on rectangles of height comparable to that
logarithm, interpolates this against $|L'/L| \ll 1$ on $\operatorname{Re} s = 3/2$; and integrating $L'/L$ from
$\operatorname{Re} s = 3/2$, where $|\log |L|| \le \log \zeta(3/2)$, gives the bound for $|L|$ and $1/|L|$. The Euler
factors at the primes dividing $q$ are at most $2$ each, which is $\ll q^{\varepsilon'}$ in all, and together
$|L(s, \psi)| \ll (q(|\operatorname{Im} s| + 2))^{5/12 + \delta' + 2\varepsilon'}$ on the new line. The decay of
$\Gamma$ makes the integral converge, and $|k_Y(\psi)| \le C Y^{1/12 - \delta'} (2q)^{5/12 + \delta' + 2\varepsilon'}$
with $C$ independent of $q$, $\psi$ and $Y$.

\section{Proofs of Lemmas~\ref{M-lem:family-two} and~\ref{M-lem:family-sharp-two}}\label{app:family}

Both lemmas are cases of two bounds for families whose polynomial part is abstract. The variable $m$ runs over a finite
set $X$ of any kind (the pairs $(m_1, m_2)$ for the two lemmas, and pairs of tuples of primes in the proof of
Lemma~\ref{M-lem:chains-two}), and it enters only through an integer value $\nu(m)$, a coefficient $c_m$, the parts
$u_j(m)$ of the comparisons' forms, the sum of the $|c_m|$, and one hypothesis: the Type~II bound at each
frequency of a separation of variables. Theorem~\ref{M-thm:type2} enters only through that hypothesis.

\subsection*{Families with an abstract polynomial part}

Fix a smooth function $f_0 \ge 0$ supported in $[1/2, 2]$, and let $F_0 = \max f_0$ and $\eta = 10^{-5}$, the largest
height exponent that Theorem~\ref{M-thm:type2} allows. Let $X$ be a finite set, and for $m \in X$ let $\nu(m) \ge 1$ be
an integer, $c_m$ a complex coefficient and $u_1(m), \dots, u_J(m)$ reals. Let $p_1, \dots, p_K$ be primes
at the scales $Q_1, \dots, Q_K > 0$, and let $C_1, \dots, C_J$ be comparisons: conditions on $(m, p)$ that hold when
$\ell_j(m, p) > 0$ and fail when $\ell_j(m, p) < 0$, where $\ell_j(m, p) = u_j(m) + \sum_i e_{ji} \log p_i$, each
with a pivot $\mathrm{piv}(j)$, an index with $e_{j, \mathrm{piv}(j)} = \pm 1$. Let $\mathcal T$ be the set of the $t =
(m, p_1, \dots, p_K)$ with $m \in X$ and each $p_i$ a prime at most $2Q_i$; for $t \in \mathcal T$ put $w(t) = c_m
\prod_i f_0(p_i/Q_i)$ and $n(t) = \nu(m) p_1 \cdots p_K$, and for a function $v$ on $\mathcal T$ put
\[
  \mathcal D(v) = \sum_{t \in \mathcal T} v(t) \, \vartheta(n(t)),
\]
with $\vartheta$ the weight of Section~\ref{M-sec:type1}. The family's sum over $\mathcal A - \mathcal B/\varphi(q)$ is
$\mathcal D(w \prod_j I_j)$, with $I_j(t) = \mathbf 1[C_j \text{ holds at } t]$. The hypothesis, for a constant $B_1 \ge
0$, is that
\begin{equation}\label{eq:H}
  \sum_{\chi \in \mathcal X^*} \Bigl|\sum_{m \in X} c'_m \, e^{2\pi i (\kappa + \sum_j \xi_j u_j(m))} \,
  \chi(\nu(m))\Bigr| \prod_i |S_{Q_i}(\chi, h_i)| \le B_1
\end{equation}
for $c' = c$ and $c' = |c|$, every $\xi \in \mathbb R^J$, every $\kappa \in \mathbb R$ and all heights $|h_i| \le
q^\eta$, with the prime sums of Theorem~\ref{M-thm:type2} for this $f_0$.

\emph{The bound with smooth weights.} Given integers $K_0 \ge 1$ and $J_0 \ge 0$ and reals $L$, $e_1 > 0$ and $E_0,
\Lambda_0 \ge 0$, there are $\delta > 0$, which depends only on $J_0$ and $e_1$, and $q_0$ such that the following holds
for every $q \ge q_0$ and every $a$ prime to $q$. Suppose that $\sum_m |c_m| \le P A_0$ for some $P > 0$ and $A_0 \ge
0$; that $1 \le K \le K_0$, $Q_i \ge q^{e_1}$ and $Q_1 \cdots Q_K \le 2^{K_0} q^L/P$; that $J \le J_0$, $|e_{ji}| \le
E_0$ and $\ell_j \le \Lambda_0 \log q$ on $\mathcal T$; and that \eqref{eq:H} holds with $B_1 = A_0 q^{L - 1/50000}$.
Then
\begin{equation}\label{eq:fam-gen}
  \Bigl|\mathcal D\Bigl(w \prod_j I_j\Bigr)\Bigr| \le \frac{A_0 \, q^{L - \delta}}{\varphi(q)}.
\end{equation}
% Lean: SieveFamilyGen.sharp_gen; its steps SieveFamilyGen.family_bound_gen (smooth family), Dsum_sharp_le_gen (sharp
% against smooth), sharp_explicit_gen, box_le_gen, pivot_le_gen

\begin{proof}
\emph{Orthogonality.} For $a$ prime to $q$, $\vartheta(n) = \varphi(q)^{-1} \sum_{\chi \in \mathcal X^*}
\overline\chi(a) \chi(n)$, so for every $v$
\begin{equation}\label{eq:orth}
  |\mathcal D(v)| \le \frac1{\varphi(q)} \sum_{\chi \in \mathcal X^*} \Bigl|\sum_{t \in \mathcal T} v(t) \chi(n(t))\Bigr|.
\end{equation}

\emph{A smooth family.} Let $g_1, \dots, g_J$ be continuous and integrable, with integrable Fourier transforms $\hat
g_j(\xi) = \int g_j(u) e^{-2\pi i u \xi} \, du$, and let $T_0 \ge 1$ with $2\pi \sum_j |e_{ji}| \, T_0 \le q^{\eta}$ for
every $i$. We show
\begin{multline}\label{eq:smooth-family}
  \sum_{\chi \in \mathcal X^*} \Bigl|\sum_{t \in \mathcal T} w(t) \prod_j g_j(\ell_j(t)) \, \chi(n(t))\Bigr|
  \le B_1 \prod_j \|\hat g_j\|_1 \\
  + \varphi(q) \, \mathfrak B \sum_j \int_{|\xi| > T_0} |\hat g_j| \prod_{l \ne j} \|\hat g_l\|_1, \qquad
  \mathfrak B = \sum_{m \in X} |c_m| \prod_i \sum_{p \le 2Q_i} f_0(p/Q_i).
\end{multline}
% Lean: SieveFamilyGen.inner_factor_gen, SieveFamilyGen.family_bound_gen, SieveSep.sum_sep, SieveSep.sep_bound
By Fourier inversion, and since the phase factorises over the variables,
\begin{gather*}
  \prod_j g_j(\ell_j) = \int_{\mathbb R^J} \prod_j \hat g_j(\xi_j) \, e^{2\pi i \sum_j \xi_j \ell_j} \, d\xi, \\
  e^{2\pi i \sum_j \xi_j \ell_j(t)} = e^{2\pi i \sum_j \xi_j u_j(m)} \prod_i p_i^{2\pi i H_i}, \qquad
  H_i = \sum_j \xi_j e_{ji}.
\end{gather*}
So the inner sum in \eqref{eq:smooth-family} is $\int \prod_j \hat g_j(\xi_j) \, W_\xi(\chi) \prod_i Q_i^{2\pi i H_i}
S_{Q_i}(\chi, h_i) \, d\xi$, where $W_\xi(\chi) = \sum_m c_m e^{2\pi i \sum_j \xi_j u_j(m)} \chi(\nu(m))$ and $h_i
= -2\pi H_i$. Take absolute values, sum over $\chi$, and bound the integrand for each $\xi$. If $|\xi_j| \le T_0$ for
every $j$, then $|h_i| \le q^{\eta}$, and \eqref{eq:H}, with $c' = c$ and $\kappa = 0$, bounds it by $B_1$. Otherwise
some $|\xi_j| > T_0$, and the trivial bound $\varphi(q) \mathfrak B$ holds. Integrating gives \eqref{eq:smooth-family}.
The same holds for a single function $g$ at one of the forms shifted by a real constant, $\ell_j + \theta$, with the
coefficients $|c_m|$ in place of $c_m$: the shift multiplies the integrand by the constant phase $e^{2\pi i \xi
\theta}$, and \eqref{eq:H} with $c' = |c|$ bounds it. This is why the hypothesis asks for $|c|$. The phase $\kappa$ of
\eqref{eq:H}, which the absolute value removes, is kept to match the formalisation's hypothesis.

\emph{The cut-off.} Let $g$ be smooth, with $g = 0$ on $(-\infty, 0]$, $g = 1$ on $[1, \infty)$ and $0 \le g \le 1$, and
let $\gamma_k = \int |g^{(k)}|$. For $0 < \theta \le 2\Lambda$ put $G_{\theta, \Lambda}(u) = g(u/\theta) - g((u -
2\Lambda)/\theta)$. It is smooth, $0 \le G_{\theta, \Lambda} \le 1$, it vanishes outside $[0, 2\Lambda + \theta]$, and
$G_{\theta, \Lambda}(u) = g(u/\theta)$ for $u \le 2\Lambda$. So if $u \le \Lambda$ and a condition holds when $u > 0$
and fails when $u < 0$,
\begin{equation}\label{eq:cutoff}
  \bigl|\mathbf 1[\text{condition}] - G_{\theta, \Lambda}(u)\bigr| \le G_{\theta, \theta}(u + \theta),
\end{equation}
since the left side vanishes unless $0 \le u < \theta$, where $G_{\theta, \theta}(u + \theta) = 1$. Integration by parts
gives $(2\pi |\xi|)^k |\hat G_{\theta, \Lambda}(\xi)| \le \|G_{\theta, \Lambda}^{(k)}\|_1 \le 2\theta^{1 - k} \gamma_k$,
and $|\hat G_{\theta, \Lambda}| \le 2\Lambda + \theta$. Hence
\begin{gather*}
  \|\hat G_{\theta, \Lambda}\|_1 \le \pi(2\Lambda + \theta) + \frac{\gamma_2}{2\pi\theta}, \\
  \int_{|\xi| > T_0} |\hat G_{\theta, \Lambda}| \le \frac{4\pi \gamma_k}{(2\pi)^k} \, \theta^{1 - k} T_0^{2 - k}
  \qquad (k \ge 2,\ T_0 \ge 1).
\end{gather*}

\emph{Sharp against smooth.} Let $\Lambda = \Lambda_0 \log q + 1$, so that $\ell_j \le \Lambda$ on $\mathcal T$, and put
$G_j = G_{\theta, \Lambda}(\ell_j)$ and $M_j = G_{\theta, \theta}(\ell_j + \theta)$. All these values lie in $[0, 1]$,
so by \eqref{eq:cutoff} $|\prod_j I_j - \prod_j G_j| \le \sum_j |I_j - G_j| \le \sum_j M_j$. Since $|\vartheta(n)| \le
\vartheta(n) + 2 \cdot \mathbf 1[(n, q) = 1]/\varphi(q)$ and $f_0 \ge 0$,
\begin{equation}\label{eq:sharp-smooth}
  \Bigl|\mathcal D\Bigl(w \prod_j I_j\Bigr)\Bigr| \le \Bigl|\mathcal D\Bigl(w \prod_j G_j\Bigr)\Bigr|
  + \sum_j \Bigl(|\mathcal D(|w| M_j)| + \frac2{\varphi(q)} \sum_{t \in \mathcal T} |w(t)| M_j(t)\Bigr).
\end{equation}
% Lean: SieveFamilyGen.Dsum_sharp_le_gen (with SieveFamily.norm_Dsum_sharp_le, SieveFamily.abs_ind_sub_G_le)
The first term and each $\mathcal D(|w| M_j)$ are smooth families: the first with $g_j = G_{\theta, \Lambda}$, the
second with the coefficients $|c_m|$ and the single function $G_{\theta, \theta}$ at the form $\ell_j + \theta$. Both
are bounded by \eqref{eq:orth} and \eqref{eq:smooth-family}. A trivial count would not do for them, since it would lose
the factor $1/\varphi(q)$ that the condition $n \equiv a$ gives. The last sum, which carries that factor already, is
counted trivially, by the pivot. Fix $m$ and the primes other than $p_{\mathrm{piv}(j)}$; then $\ell_j$ is a constant
plus or minus $\log p_{\mathrm{piv}(j)}$, and $M_j \ne 0$ forces $-\theta \le \ell_j \le 2\theta$, which confines
$p_{\mathrm{piv}(j)}$ to an interval $[y, e^{3\theta} y]$. At most $(e^{3\theta} - 1) 2Q_{\mathrm{piv}(j)} + 1$ integers
up to $2Q_{\mathrm{piv}(j)}$ lie in it, so
\[
  \sum_{t \in \mathcal T} |w(t)| M_j(t)
  \le F_0 \bigl((e^{3\theta} - 1) 2Q_{\mathrm{piv}(j)} + 1\bigr) \sum_m |c_m| \prod_{i \ne \mathrm{piv}(j)} \sum_{p \le
  2Q_i} f_0(p/Q_i).
\]

\emph{The parameters.} Put $\varepsilon_2 = 1/(400000(J_0 + 1))$, $\theta = q^{-\varepsilon_2}$, $T_0 = q^{1/200000}$,
$k = 1200003$ and $\delta = \min(\varepsilon_2, e_1)/2$. For large $q$:
\begin{itemize}
\item $2\pi J_0 E_0 T_0 \le q^{\eta}$, so \eqref{eq:smooth-family} applies, as $\sum_j |e_{ji}| \le J_0 E_0$;
\item $\pi(2\Lambda + 1) \le q^{\varepsilon_2}$, so $\|\hat G_{\theta, \Lambda}\|_1$ and $\|\hat G_{\theta, \theta}\|_1$
  are at most $C q^{\varepsilon_2}$ with $C = 1 + \gamma_2/(2\pi)$, and a product of at most $J_0$ of them is at most
  $C^{J_0 + 1} q^{1/400000}$;
\item $\theta^{1 - k} T_0^{2 - k} = q^{\varepsilon_2(k - 1) - (k - 2)/200000} \le q^{-3}$, which bounds the tails.
\end{itemize}
The constants come from $\eta = 10^{-5}$: the norms lose at most $q^{\eta/4}$, $T_0 = q^{\eta/2}$ leaves room for the
factor $2\pi J_0 E_0$ below $q^{\eta}$, and $k = 1200003$ is the least $k$ with $(k - 1)\eta/4 - (k - 2)\eta/2 \le -3$,
so that the tails are at most $q^{-3}$ whatever $J_0$. Moreover $\mathfrak B \le P A_0 (3F_0)^K \prod_i Q_i \le A_0
(3F_0 + 1)^{K_0} 2^{K_0} q^L$, since at most $3Q$ integers lie in $[1, 2Q]$ for $Q \ge 1$. Finally $e^{3\theta} - 1 \le
6\theta$ for $\theta \le 1/3$ and $Q_{\mathrm{piv}(j)} \ge q^{e_1}$ give $(12\theta Q_{\mathrm{piv}(j)} + 1) \prod_{i
\ne \mathrm{piv}(j)} Q_i \le (12\theta + q^{-e_1}) \prod_i Q_i$. Putting these into \eqref{eq:sharp-smooth}, with
\eqref{eq:orth} and \eqref{eq:smooth-family} at $B_1 = A_0 q^{L - 1/50000}$, and using $\varphi(q) \le q$, the sum is at
most
\[
  \frac{A_0 q^L}{\varphi(q)} \bigl(a_1 q^{-7/400000} + a_2 q^{-1} + a_3(12 q^{-\varepsilon_2} + q^{-e_1})\bigr),
\]
where $a_1$, $a_2$ and $a_3$ depend only on $K_0$, $J_0$ and $F_0$. The three terms come from the smooth families, from
the tails and from the pivot counts. Each power of $q$ in the bracket is at most $q^{-2\delta}$, so for large $q$ the
sum is at most $A_0 q^{L - \delta}/\varphi(q)$. The hypothesis \eqref{eq:H} was used once for the smoothed family and
once for each near-boundary family.
\end{proof}

\emph{The bound without the weights.} Take for $f_0$ the fixed function $f_0(y) = F(y) - F(y/2)$, with $F$ smooth and
nondecreasing, $0$ on $(-\infty, 1/2]$ and $1$ on $[1, \infty)$. It is supported in $[1/2, 2]$, takes values in $[0,
1]$, and $\sum_{0 \le t \le t_*} f_0(y/2^t) = F(y) - F(y/2^{t_* + 1}) = 1$ for $1 \le y \le 2^{t_*}$. Given integers
$K_0 \ge 1$, $J_0 \ge 0$ and $c_0 \ge 0$ and reals $0 < L \le 5/2$, $e_1 > 0$ and $E_0, \Lambda_1 \ge 0$, there are
$\delta > 0$ and $q_0$ such that the following holds for every $q \ge q_0$ and every $a$ prime to $q$. Let $X$ be cut
into blocks $X_b$, labelled by at most $(3 + 4 \log q)^{c_0}$ labels $b$, each with a size $P_b > 0$ such that $\nu(m) >
P_b$ on $X_b$ and $\sum_{m \in X_b} |c_m| \le P_b A_0$, for some $A_0 \ge 0$. Let $1 \le K \le K_0$ primes $p_1, \dots,
p_K$ run over all the primes up to $N_2$, and let $J \le J_0$ comparisons as above, with $|e_{ji}| \le E_0$,
$u_j(m) \le \Lambda_1 \log q$ on $X$ and pivots, cut out the $(m, p)$; suppose that wherever they all hold, every
$p_i \ge 2q^{e_1}$ and $\nu(m) p_1 \cdots p_K \le 2q^L$. If \eqref{eq:H} holds on each block $X_b$ in place of $X$, with
$B_1 = A_0 q^{L - 1/50000}$, for all scales $Q_i \ge q^{e_1}$ with $Q_1 \cdots Q_K \le 2^{K_0 + 1} q^L/P_b$, then
\begin{equation}\label{eq:fam-unw}
  \Bigl|\sum_{m \in X} \sum_{p_1, \dots, p_K \le N_2} c_m \prod_j I_j(m, p) \, \vartheta(\nu(m) p_1 \cdots p_K)\Bigr|
  \le \frac{A_0 \, q^{L - \delta}}{\varphi(q)}.
\end{equation}
% Lean: SieveUnwGen.unw_gen (with SieveUnwGen.Dsum_fiber_gen, SieveUnwGen.box_change_gen, SieveDyadic.sum_bump)

\begin{proof}
Let $t_* = \lfloor \log_2 N_2 \rfloor + 1$, so that every prime up to $N_2$ is below $2^{t_*}$, and $t_* + 1 \le 3 + 4
\log q$ since $L \le 5/2$. Split the sum over the blocks, and in each block insert $\sum_{0 \le t_i \le t_*}
f_0(p_i/2^{t_i}) = 1$ for each prime. This writes the sum as a sum of at most $(3 + 4 \log q)^{c_0 + K_0}$ pieces, one
for each block and each choice of the scales $Q_i = 2^{t_i}$, in which $m$ runs over the block and the primes carry the
weights $f_0(p_i/Q_i)$. In a piece the primes may run over the ranges $p_i \le 2Q_i$ of a family instead of the primes
up to $N_2$: the weights vanish for $p_i > 2Q_i$, and a term with some $p_i > N_2$ violates the comparisons, which force
$\nu(m) \prod_i p_i \le 2q^L$. On a piece's box $p_i \le 2Q_i \le 8q^L$, so every form is at most $(\Lambda_1 + K_0 E_0
(L + 3)) \log q$ for $q \ge 2$. A piece with some $Q_i < q^{e_1}$ vanishes, since its weights force $p_i < 2q^{e_1}$;
and so does a piece with $\prod_i Q_i > 2^{K_0 + 1} q^L/P_b$, since its weights force $p_i \ge Q_i/2$, and so $\prod_i
Q_i \le 2^K \prod_i p_i \le 2^{K + 1} q^L/\nu(m) < 2^{K_0 + 1} q^L/P_b$ wherever the comparisons hold. Every other piece
is a family of the bound with smooth weights, with $K_0 + 1$ in place of $K_0$, $F_0 = 1$, $P = P_b$ and $\Lambda_0 =
\Lambda_1 + K_0 E_0 (L + 3)$, whose hypothesis \eqref{eq:H} is the one assumed on its block; so \eqref{eq:fam-gen}
bounds it by $A_0 q^{L - \delta_1}/\varphi(q)$. For large $q$ the number of pieces is at most $q^{\delta_1/2}$, and
$\delta = \delta_1/2$ will do.
\end{proof}

\subsection*{Two variables}

Let $S_1 \subseteq (P_1, 2P_1]$ and $S_2 \subseteq (P_2, 2P_2]$ be sets of integers, with $(\log_q P_1, \log_q P_2) \in
\mathcal R$, and let $a'$ and $b'$ be coefficients with $|a'_{m_1}| \le A_0$ and $|b'_{m_2}| \le B_0$. For reals
$u_j(m_1)$, $v_j(m_2)$, $\xi_j$ and $\kappa$, the phase $\kappa + \sum_j \xi_j(u_j(m_1) + v_j(m_2))$ is
a phase on $m_1$ plus a phase on $m_2$, and $\chi(m_1 m_2) = \chi(m_1) \chi(m_2)$, so the twisted sum over the pairs is
a product of two Dirichlet polynomials:
\begin{gather*}
  \sum_{m_1 \in S_1} \sum_{m_2 \in S_2} a'_{m_1} b'_{m_2} \, e^{2\pi i(\kappa + \sum_j \xi_j(u_j(m_1) +
  v_j(m_2)))} \chi(m_1 m_2) = W_1(\chi) W_2(\chi), \\
  W_1(\chi) = \sum_{m_1 \in S_1} a'_{m_1} e^{2\pi i(\kappa + \sum_j \xi_j u_j(m_1))} \chi(m_1), \\
  W_2(\chi) = \sum_{m_2 \in S_2} b'_{m_2} e^{2\pi i \sum_j \xi_j v_j(m_2)} \chi(m_2):
\end{gather*}
polynomials at the heights $0$ whose coefficients have the moduli $|a'_{m_1}|$ and $|b'_{m_2}|$. Since $S_\iota$ has at
most $P_\iota$ elements, Theorem~\ref{M-thm:type2}, with this $\eta$, gives for $q$ beyond its $q_0$
\begin{multline}\label{eq:pair}
  \sum_{\chi \in \mathcal X^*} |W_1(\chi) W_2(\chi)| \prod_i |S_{Q_i}(\chi, h_i)| \\
  \le q^{L - 1/50000} \Bigl(\frac1{P_1} \sum_{S_1} |a'|^2\Bigr)^{1/2} \Bigl(\frac1{P_2} \sum_{S_2} |b'|^2\Bigr)^{1/2}
  \le A_0 B_0 \, q^{L - 1/50000}
\end{multline}
whenever $K \le K_0$, $Q_i \ge q^{e_1}$, $|h_i| \le q^\eta$ and $Q_1 \cdots Q_K \le 2^{K_0} q^L/(P_1 P_2)$.
% Lean: SieveFamilyTwo.pair_sum_eq, SieveFamilyTwo.box_two

\begin{proof}[Proof of Lemma~\ref{M-lem:family-two}]
% Lean: SieveFamilyTwo.typeII_sharp_two
Take for $X$ the pairs $(m_1, m_2) \in S_1 \times S_2$, with the values $m_1 m_2$, the coefficients $a_{m_1} b_{m_2}$
and the parts $u_j(m_1) + v_j(m_2)$ of the forms. Then $\sum |a_{m_1} b_{m_2}| \le P_1 P_2 A_0 B_0$, and the
moduli $|a_{m_1} b_{m_2}| = |a_{m_1}| \, |b_{m_2}|$ factorise as well, so \eqref{eq:pair}, with $a' = a$ or $|a|$ and
$b' = b$ or $|b|$, is the hypothesis \eqref{eq:H} with $B_1 = A_0 B_0 q^{L - 1/50000}$. The bound with smooth weights,
with $P = P_1 P_2$ and $A_0 B_0$ in place of $A_0$, is the lemma; its $\delta$ depends only on $J_0$ and $e_1$.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{M-lem:family-sharp-two}]
% Lean: SieveUnwTwo.typeII_unw_two
Take $X$ as in the previous proof, with the function $f_0$ of the bound without the weights, and cut it into the blocks
of the pairs with $P_1 2^{b_1} < m_1 \le P_1 2^{b_1 + 1}$ and $P_2 2^{b_2} < m_2 \le P_2 2^{b_2 + 1}$. Since $m_\iota
\le 2q^L$ and $L \le 5/2$, each $b_\iota$ takes at most $3 + 4 \log q$ values, so there are at most $(3 + 4 \log q)^2$
blocks. The block $b = (b_1, b_2)$ has the size $P_b = P_1 2^{b_1} P_2 2^{b_2}$, below $m_1 m_2$, and at most $P_b$
pairs, so the sum of $|a_{m_1} b_{m_2}|$ over it is at most $P_b A_0 B_0$. A block that holds a pair has $P_\iota \le
P_\iota 2^{b_\iota} < m_\iota \le P_\iota'$, so the exponents of its lower ends $(P_1 2^{b_1}, P_2 2^{b_2})$ lie in the
rectangle, where $\mathcal R$ holds; and \eqref{eq:pair} at these lower ends, with $K_0 + 1$ in place of $K_0$ in
Theorem~\ref{M-thm:type2}, is \eqref{eq:H} on the block for all scales with $Q_1 \cdots Q_K \le 2^{K_0 + 1} q^L/P_b$. A
block that holds no pair contributes nothing. The bound without the weights, with $c_0 = 2$ and $A_0 B_0$ in place of
$A_0$, is the lemma.
\end{proof}

The proof of Lemma~\ref{M-lem:chains-two} applies the bound without the weights directly. There $X$ is the set of the
pairs of the parts, on $\mathcal S_1$ and on $\mathcal S_2$, of the tuples of primes, with the product of their entries
as the value and the coefficient $1$, cut into the pairs of dyadic blocks of the two products; the comparisons with a
prime of the rest are the pivoted ones, and those with none are conditions between the two parts, kept in the kernel of
the sum over the pairs. On a block, \eqref{eq:H} follows from \eqref{eq:pair} and Lemma~\ref{M-lem:cross}: a polynomial
over the parts is one over their products, with coefficients at most the number $(K + 1)^K$ of tuples with a given
product, and the conditions in the kernel cost a power of $\log q$; both go into $A_0$, which is a power of $\log q$.

\section{Proofs of Propositions~\ref{M-prop:harman-two} and~\ref{M-prop:harman-fill}, and of
  Lemma~\ref{M-lem:leaves}}\label{app:harman}

Write $\mathcal P[y, z)$ for the set of the primes in $[y, z)$ and $\mathcal P(y, z)$ for the set of those in $(y, z)$;
say that $n$ is \emph{$z$-rough} if it has no prime factor below $z$, and let $R_z(n)$ be $1$ if $n$ is $z$-rough and
$0$ otherwise. Throughout, $x = q^L$, $N_1 = \lfloor x \rfloor$ and $N_2 = \lfloor 2x \rfloor$, the tuples $(r_1, \dots,
r_j)$ are ordered, $r = r_1 \cdots r_j$, and $\prod P$ is the product of the elements of a set $P$ of primes. For
$\mathcal C = \mathcal A$ or $\mathcal B$ and every $d \ge 1$, $S(\mathcal C_d, z)$ is the number of $z$-rough multiples
of $d$ in $\mathcal C$, so that
\begin{equation}\label{eq:SAB}
  S(\mathcal A_d, z) - \frac{S(\mathcal B_d, z)}{\varphi(q)} = \sum_{N_1 < n \le N_2} \vartheta(n) \, \mathbf 1[d \mid n]
  \, R_z(n).
\end{equation}
The first two subsections hold the tree identity and the bound at a leaf, which the proofs of the two propositions
share, and the third the spline weights of Section~\ref{M-sec:type1}, which the proof of Lemma~\ref{M-lem:leaves} uses.

\subsection*{Buchstab's tree with a stopping rule}

Let $1 \le w \le z$, let $d_0$ be a positive integer whose prime factors are all at least $z$, and let $\gamma$ and $E$
be reals with $0 \le \gamma < E$. For a set $P$ of primes put $d_P = d_0 \prod P$ and $\gamma_P = \gamma \prod P$. Then
for every $n$
\begin{multline}\label{eq:tree}
  \mathbf 1[d_0 \mid n] \, R_z(n)
  = \sum_{P \subseteq \mathcal P[w, z)} (-1)^{|P|} \, \mathbf 1[\gamma_P < E] \, \mathbf 1[d_P \mid n] \, R_w(n) \\
  + \sum_{p \in \mathcal P[w, z)} \ \sum_{P \subseteq \mathcal P(p, z)} (-1)^{|P| + 1} \,
    \mathbf 1[\gamma_P < E \le \gamma_P \, p] \, \mathbf 1[p \, d_P \mid n] \, R_p(n).
\end{multline}
% Lean: SieveTreeStop.tree_weighted_stop, SieveP6Tree.tree_split, SieveP6Tree.tree_chains (z = z_0);
% SieveFillTree.tree_split_below, SieveFillTree.tree_last (z the last prime of a chain, gamma = 1)
This is what iterating Buchstab's identity $S(\mathcal C_m, z) = S(\mathcal C_m, w) - \sum_{w \le p < z} S(\mathcal
C_{mp}, p)$ from $d_0$ at $z$ gives, when a branch $m = d_P$ is stopped as soon as $\gamma_P$ reaches $E$: the first sum
collects the \emph{leaves} $S(\mathcal C_d, w)$, $d = d_P$, and the second the \emph{stops} $S(\mathcal C_W, p)$, $W = p
\, d_P$, whose last prime $p$ is the least. It is proved directly, for each $n$. If $d_0 \nmid n$ or $n$ is not
$w$-rough, every term vanishes. Otherwise let $Q$ be the set of the primes of $[w, z)$ that divide $n$; the left side is
$\mathbf 1[Q = \emptyset]$. Since the primes of $d_0$ are at least $z$, a leaf $P$ counts $n$ exactly when $P \subseteq
Q$, and a stop $(p, P)$ exactly when $p = \min Q$ and $P \subseteq Q \setminus \{p\}$. If $Q = \emptyset$, only the leaf
$P = \emptyset$ counts $n$, with $\mathbf 1[\gamma < E] = 1$. If $Q \ne \emptyset$, let $p_0 = \min Q$; grouping $P
\subseteq Q \setminus \{p_0\}$ with $P \cup \{p_0\}$, the right side is
\[
  \sum_{P \subseteq Q \setminus \{p_0\}} (-1)^{|P|} \bigl(\mathbf 1[\gamma_P < E] - \mathbf 1[\gamma_P \, p_0 < E]
  - \mathbf 1[\gamma_P < E \le \gamma_P \, p_0]\bigr) = 0,
\]
since $\gamma_P \le \gamma_P \, p_0$. Multiplying \eqref{eq:tree} by $\vartheta(n)$ and summing over $n \in (N_1, N_2]$,
\eqref{eq:SAB} writes $S(\mathcal A_{d_0}, z) - S(\mathcal B_{d_0}, z)/\varphi(q)$ as the sum over the leaves of
$(-1)^{|P|}(S(\mathcal A_d, w) - S(\mathcal B_d, w)/\varphi(q))$ plus the sum over the stops of $(-1)^{|P| +
1}(S(\mathcal A_W, p) - S(\mathcal B_W, p)/\varphi(q))$. The proofs below write a set $P$ as the decreasing tuple $y_1 >
\dots > y_s$ of its elements; with the primes of $d_0$ in front, the leaves and the stops are then chains of primes.

\subsection*{A leaf by Brun's sieve}

Let $C_0 \ge 1$ be an absolute constant with $\sum_{p \in \mathcal P[y, y^2)} 1/p \le C_0$ for every $y \ge 1$; it
exists by Mertens' theorem (the library takes it from the prime number theorem). The number $\zeta_k$ of
Section~\ref{M-sec:type1} is
\[
  \zeta_k = e^{2C_0} \cdot \frac{2 C_0^k}{k!} \cdot \exp\Bigl(e^{2C_0} \frac{2 C_0^{k + 1}}{(k + 1)!}\Bigr),
\]
and $(2k + 4)^3 \zeta_k \to 0$ as $k \to \infty$, since the last factor tends to $1$ and $C_0^k/k!$ decays faster than
any power of $1/k$. Let $k \ge 2$ be even with $2C_0^2 \le (k + 1)(k + 2)$, let $w \ge 2$, and let $d$ have all its
prime factors at least $w$. We show
\begin{equation}\label{eq:leaf}
  \Bigl|S(\mathcal A_d, w) - \frac{S(\mathcal B_d, w)}{\varphi(q)}\Bigr|
  \le \frac{2 \zeta_k (N_2 - N_1)}{d \varphi(q) \log w} + 24 \, w^{2k + 7}.
\end{equation}
% Lean: SieveProp6.leaf_bound; SieveBrun.fundamental (Brun's sieve), SieveLeaf.leaf_gap (the gap),
% SieveLeaf.prod_T_le, SieveLeaf.prod_primesBelow_le (the main term)
If $(d, q) > 1$ both counts vanish, so let $(d, q) = 1$. Sieve by the primes below $w$, in the ranges $T_i = \mathcal
P[w^{2^{-i - 1}}, w^{2^{-i}})$ for $0 \le i < J_w = 2 \lfloor \log_2 w \rfloor + 2$, which cover them, at the even
levels $k_i = k + 2i$, which grow with $i$ so that the errors $C_0^{k_i}/k_i!$ below have a sum that does not depend on
$J_w$. Brun's sieve \cite{brun}, in the following form, is used, for disjoint sets $T_i$ of primes and even levels $k_i
\ge 2$. Let $\mathcal C$ be a finite set of integers, $Z \ge 0$, and $g$ a function on the primes of $\bigcup T_i$ with
$0 \le g \le 1$. Suppose that for every choice of $s_i \subseteq T_i$, with $\Pi = \prod_i \prod_{p \in s_i} p$, the
count $\#\{n \in \mathcal C : \Pi \mid n\}$ is $Z \prod_i \prod_{p \in s_i} g(p)$ within $\Delta$. Then $\#\{n \in
\mathcal C : (n, \prod_i \prod_{T_i} p) = 1\}$ is $Z \prod_i \prod_{p \in T_i} (1 - g(p))$ within
\[
  Z \sum_i e_{k_i}(T_i) \prod_{l \ne i} F_{k_l}(T_l) + (J_w + 1) \, \Delta \prod_i \sum_{m \le k_i} \binom{|T_i|}{m},
\]
where $e_m(T) = \sum_{s \subseteq T,\ |s| = m} \prod_{p \in s} g(p)$ and $F_k(T) = \sum_{m \le k} (-1)^m e_m(T)$.
Bonferroni's inequalities say that $F_{k - 1}(T) \le \prod_{p \in T}(1 - g(p)) \le F_k(T)$ for every even $k$, and the
same for any weights in $[0, 1]$ in place of $g$. Take them for each $n \in \mathcal C$ with the weights $\mathbf 1[p
\mid n]$, and use the elementary inequality $\prod_i \mathrm{hi}_i - \sum_i (\mathrm{hi}_i - \mathrm{lo}_i) \prod_{l \ne
i} \mathrm{hi}_l \le \prod_i \varrho_i$ for $\varrho_i \ge 0$ and $\mathrm{lo}_i \le \varrho_i \le \mathrm{hi}_i$ (the
$\mathrm{lo}_i$, odd truncations, may be negative). Summed over $n$, and expanded by the hypothesis, they place the
count, up to $(J_w + 1)$ times $\Delta \prod_i \sum_{m \le k_i} \binom{|T_i|}{m}$, in the interval from $Z \prod_i
F_{k_i} - Z \sum_i e_{k_i} \prod_{l \ne i} F_{k_l}$ to $Z \prod_i F_{k_i}$. The model $Z \prod_i V_i$, with $V_i =
\prod_{p \in T_i}(1 - g(p))$, lies in the same interval: Bonferroni's inequalities for $g$ give $F_{k_i} - e_{k_i} =
F_{k_i - 1} \le V_i \le F_{k_i}$, hence $0 \le \prod_i F_{k_i} - \prod_i V_i \le \sum_i e_{k_i} \prod_{l \ne i}
F_{k_l}$.

Take $g(p) = 1/p$ for $p \nmid q$ and $g(p) = 0$ for $p \mid q$. For $\mathcal A_d$, take $\mathcal C = \{n \in (N_1,
N_2] : n \equiv a,\ d \mid n\}$, $Z = (N_2 - N_1)/(dq)$ and $\Delta = 1$: if no prime of $\Pi$ divides $q$,
Lemma~\ref{M-lem:type1} with $M = d\Pi$ gives the count within $1$, and otherwise the count and the model both vanish.
For $\mathcal B_d$, take $\mathcal C = \{n \in (N_1, N_2] : (n, q) = 1,\ d \mid n\}$, $Z\varphi(q)$ and $\Delta =
\varphi(q)$, and divide by $\varphi(q)$. Both main terms are then $ZV$ with $V = \prod_{p < w,\ p \nmid q}(1 - 1/p)$, so
that
\begin{gather*}
  \Bigl|S(\mathcal A_d, w) - \frac{S(\mathcal B_d, w)}{\varphi(q)}\Bigr| \le 2(ZG + \mathrm{rem}), \\
  G = \sum_i e_{k_i}(T_i) \prod_{l \ne i} F_{k_l}(T_l), \qquad
  \mathrm{rem} = (J_w + 1) \prod_i \sum_{m \le k_i} \binom{|T_i|}{m}.
\end{gather*}
Each $T_i$ lies in some $\mathcal P[y, y^2)$, so $\sum_{p \in T_i} g(p) \le C_0$ and, since $1 - t \ge e^{-2t}$ for $0
\le t \le 1/2$, $V_i \ge e^{-2C_0}$. With $e_m \le C_0^m/m!$, $F_{k_l} \le V_l + e_{k_l + 1} \le V_l (1 + e^{2C_0}
C_0^{k_l + 1}/(k_l + 1)!)$ and $\prod_{l \ne i} V_l \le e^{2C_0} V$, this gives $G \le V e^{2C_0} \sum_i
\frac{C_0^{k_i}}{k_i!} \prod_l \bigl(1 + e^{2C_0} \frac{C_0^{k_l + 1}}{(k_l + 1)!}\bigr)$. Since $2C_0^2 \le (k + 1)(k +
2)$, the terms of $\sum_i C_0^{k + 2i}/(k + 2i)!$ decrease at least geometrically with ratio $1/2$, so the sum is at
most $2C_0^k/k!$; likewise, since $2C_0^2 \le (k + 2)(k + 3)$, $\sum_l C_0^{k_l + 1}/(k_l + 1)! \le 2C_0^{k + 1}/(k +
1)!$; and with $\prod (1 + t_l) \le \exp \sum t_l$ we get $G \le \zeta_k V$. So the difference is at most $2\zeta_k ZV +
2\,\mathrm{rem}$. Next, $V \le q/(\varphi(q) \log w)$. Indeed $V \prod_{p < w,\ p \mid q}(1 - 1/p) = \prod_{p < w}(1 -
1/p)$, the second product on the left is at least $\prod_{p \mid q}(1 - 1/p) = \varphi(q)/q$, and $\prod_{p < w}(1 -
1/p) \le 1/\log w$, since $\prod_{p < w}(1 - 1/p)^{-1}$ is the sum of $1/m$ over the integers $m \ge 1$ whose prime
factors are all below $w$, which is at least $\sum_{m < w} 1/m \ge \log w$. So $ZV \le (N_2 - N_1)/(d \varphi(q) \log
w)$. Finally $\sum_{m \le k_i} \binom{|T_i|}{m} \le 2 \max(1, |T_i|)^{k_i}$ with $|T_i| \le w^{2^{-i}}$, $\sum_i
2^{-i}(k + 2i) \le 2k + 4$, $J_w + 1 \le 3w$ and $2^{J_w} \le 4w^2$ give $\mathrm{rem} \le 12 w^{2k + 7}$.

\subsection*{The spline weights of Section~\ref{M-sec:type1}}

For integers $A < B$, $\ell \ge 1$ and $r \ge 1$ let $w(k)$ be $\ell^{-r}$ times the number of $(i, j_1, \dots, j_r)$
with $A < i \le B$, $0 \le j_\nu < \ell$ and $i + j_1 + \cdots + j_r = k$. It is the average over the $\ell^r$ tuples
$(j_\nu)$ of the indicators of the intervals $(A + \sum j_\nu, B + \sum j_\nu]$, so $0 \le w \le 1$, $w = 1$ on $(A +
r(\ell - 1), B]$, $w = 0$ off $(A, B + r(\ell - 1)]$, and $\sum_k w(k) = B - A$. Its transform factors: $\hat w(h) =
\ell^{-r} \bigl(\sum_{A < i \le B} e_q(hi)\bigr) \bigl(\sum_{j < \ell} e_q(hj)\bigr)^r$. For $h$ the residue of an
integer $t$ with $0 < |t| \le q/2$, a geometric sum at $e_q(t)$, of any length, is at most $q/(2|t|)$ in size, so $|\hat
w(h)| \le (B - A)(q/(2\ell|t|))^r$, which is at most $(B - A)(q/(2\ell(H + 1)))^r$ when $|t| > H$. If each residue class
mod $q$ holds at most $M$ elements of the support, Parseval's identity and $|w(k) w(k')| \le \tfrac12(|w(k)|^2 +
|w(k')|^2)$ over the pairs in a class give $\sum_h |\hat w(h)|^2 \le qM \sum_k |w(k)|^2$; for the support, an interval
of length $n = B - A + r(\ell - 1)$, $M = \lfloor n/q \rfloor + 1$. Finally $|E_w| \le d(q)$: for the indicator of an
interval $(A', B']$, Möbius inversion over the divisors of $q$ writes $E_w$ as $\sum_{d \mid q} \mu(d)\bigl((\lfloor
B'/d \rfloor - \lfloor A'/d \rfloor) - (B' - A')/d\bigr)$, and each bracket is less than $1$ in size; the spline
averages such intervals.

\subsection*{Proof of Lemma~\ref{M-lem:leaves}}
% Lean: SieveP6LeavesWide.leaves_chain_wide, SieveP6LeavesWide.leavesOK_wide, SieveTypeIBAsymWide.leaves_ib_wide

\begin{proof}
Write $R = R_w$, $\mathrm{rng}(d, m)$ for the indicator of $N_1 < dm \le N_2$, and $I(\mathbf y_1, \mathbf y_2)$ for
that of the comparisons. For $d$ with no prime factor below $w$, by \eqref{eq:SAB}, $S(\mathcal A_d, w) - S(\mathcal
B_d, w)/\varphi(q) = \sum_m \mathrm{rng}(d, m) R(m) \, \vartheta(dm)$. The proof has four steps: Brun's weights for $R$,
a sandwich of the cofactor's range by splines, blocks of the moduli, and Lemma~\ref{M-lem:typeIB} on each pair of
blocks. The wider region comes from the last: the frequencies are cut at $H = \lceil qY/\ell \rceil$, and $2HD$ may
exceed $q$.

\emph{Brun's weights.} Sift by the primes below $w$ in the ranges $T_i$ of the proof of \eqref{eq:leaf}, at the even
levels $k_i = k + 2i$. Bonferroni's inequalities give weights $\lambda^+$ above and $\lambda^-$ below $R$: $\sum_\tau
\lambda^-_\tau \mathbf 1[f_\tau \mid m] \le R(m) \le \sum_\sigma \lambda^+_\sigma \mathbf 1[f_\sigma \mid m]$, with
moduli $f \le w^{2k + 4}$, $\sum |\lambda| \le 2\,\mathrm{rem}$ and $\sum |\lambda|/f$ at most a power of $\log q$; and
the gap $\Gamma$ between the two weights' main terms, $\sum_\sigma \lambda^+_\sigma \mathbf 1[(f_\sigma, q) =
1]/f_\sigma - \sum_\tau \lambda^-_\tau \mathbf 1[(f_\tau, q) = 1]/f_\tau$, is at most $\zeta_k q/(\varphi(q) \log w)$ in
size, as the gap $G$ of that proof.

\emph{The sandwich.} Split each $\lambda$ into its positive and negative parts. With splines $w_- \le \mathrm{rng}(d
f_\sigma, \cdot) \le w_+$, chosen below for each block, the count of $\mathcal A$ is at most $\sum_\sigma
(\lambda^+_\sigma)_+ \sum_k w_+(k) \mathbf 1[d f_\sigma k \equiv a] - \sum_\sigma (\lambda^+_\sigma)_- \sum_k w_-(k)
\mathbf 1[d f_\sigma k \equiv a]$, and at least the same expression with the weights $\lambda^-$ and the splines
exchanged; and the same holds for $\mathcal B/\varphi(q)$. The difference between the $\mathcal A$- and the $\mathcal
B/\varphi(q)$-counts is thus bounded by two kinds of term: the difference of the main terms, which carries the gap
$\Gamma$ and the difference of the splines' masses, and bilinear sums $\sum \lambda \sum_k w(k) \, \vartheta(d f k)$.

\emph{The parameters.} Put $X_1 = q^{L - 1 - \xi}$, $X_2 = q^{L + c}$ and $X_3 = q^{L - 1 - \delta'}$, so that every leaf
has $d_1 f \le X_1$, $d_1 f d_2^2 \le X_2$ and $d_1 d_2 \ge X_3$ for every modulus $f$ of Brun's weights. Let $s =
\min(\kappa/(4(J_0 + 6)), \varepsilon_w, 1/(4(J_0 + 6)))$, $\theta = q^{-s}$, $\rho = 1 + \theta$, $y = \min(\kappa/4,
1/8)$, $Y = q^y$, $r = \lceil 2/y \rceil$, and $F = \lfloor C_d q^s \rfloor$, where $d(n) \le C_d n^{s/(L + 2)}$ for
every $n \ge 1$, so that $F$ bounds the number of divisors of $q$ and of every integer up to $q^{L + 2}$.

\emph{Blocks.} Group the pairs $(d_1 f, d_2)$, $d_1 = \prod \mathbf y_1$ and $d_2 = \prod \mathbf y_2$, by the blocks
$\lfloor \log(d_1 f)/\log \rho \rfloor$ and $\lfloor \log d_2/\log \rho \rfloor$; a block pair $(\beta_1, \beta_2)$ puts
$d_1 f d_2$ in $[\rho^{\beta_1 + \beta_2}, \rho^{\beta_1 + \beta_2 + 2})$. At the scale $D' = \rho^{\beta_1 + \beta_2}$
one majorant $w_+$, equal to $1$ on $(\lfloor N_1/(D'\rho^2) \rfloor, \lfloor N_2/D' \rfloor]$, and one minorant $w_-$,
supported in $(\lfloor N_1/D' \rfloor, \lfloor N_2/(D'\rho^2) \rfloor]$, both splines of step $\ell = \lfloor \theta
N_1/(rD'\rho^2) \rfloor + 1$ and power $r$, sandwich $\mathrm{rng}(d, \cdot)$ for every $d$ of the block pair, and their
masses differ by at most $(N_1 + N_2)(1 - \rho^{-2})/D' + 2\theta N_1/(D'\rho^2) + 2$. Since $d_1 f \le X_1$ and $d_2
\le X_2^{1/2}$, at most $(3Lq^s \log q)^2$ block pairs hold a leaf.

\emph{A block pair.} On a block pair Lemma~\ref{M-lem:typeIB} applies, at each frequency of the separation below, with
$\mathcal D_1$ the values $d_1 f$, whose coefficients aggregate the fibres $\{(\mathbf y_1, \sigma) : \prod \mathbf y_1
\cdot f_\sigma = d_1 f\}$ (at most $(J_w + 1) d(d_1 f)$ elements, by the uniqueness of the factorisation), $\mathcal
D_2$ the values $d_2$ (a decreasing tuple of primes is determined by its product), $P_\iota = \rho^{\beta_\iota + 1}$,
$D = \lfloor P_2 \rfloor$, and the cut $H = \lceil qY/\ell \rceil$; each residue class contains the inverses of at most
$\lfloor P_1/q \rfloor + 1$ of the values $d_1 f$, since those with a given inverse lie in one residue class mod $q$.
The comparisons are replaced by smooth cut-offs of their forms, as in \S\ref{app:family}: the smooth part is separated
by Fourier inversion into bilinear forms of the lemma's kind, at a cost of the cut-offs' Fourier masses, and the part
near the boundaries is treated as there: with $|\vartheta(n)| \le \vartheta(n) + 2 \cdot \mathbf 1[(n, q) =
1]/\varphi(q)$, the part with $\vartheta$ is again a bilinear form of the lemma's kind, with the narrow cut-offs, and
the part with $2/\varphi(q)$ is counted along the pivots. Provided $2H < q$ and $F$ bounds the number of divisors of the
integers up to $HD$, a block pair is so bounded by its Fourier masses and fibre sizes times
\begin{multline}\label{eq:block-pair}
  \sqrt{P_1 P_2 (P_1/q + 1) F (2HP_2/q + 1) (n/q + 1) n} \\
  + (B - A) \Bigl(\frac q{2\ell(H + 1)}\Bigr)^r P_1 P_2 + P_1 P_2 \, \frac{d(q)}{\varphi(q)},
\end{multline}
with $n$ the length and $B - A$ the mass of the spline, plus the counted parts near the boundaries.
% Lean: SieveTypeIBPairWide.pair_le_wide (typeIB_smooth_wide, typeIB_sharp0_wide), SieveTypeIB.typeIB_bound_wide,
% SieveSpline.typeIB_spline_wide

\emph{The cut.} On a block pair that holds a leaf, $D' \le d_1 f d_2 < D'\rho^2$ and $(d_1 f d_2)^2 \le d_1 f \cdot d_1
f d_2^2 \le X_1 X_2 \le q^{2L - 1}$, since $c \le \xi$. So $\theta N_1/(rD'\rho^2) \le \theta N_1/(rX_3) \le q^{1 +
\delta' - s}$, and the step has $\ell \le q^{1 + \delta' - s} + 1 \le qY$ for large $q$, since $\delta' \le y$: this is
where $\delta' \le \min(\kappa/4, 1/8)$ enters. Hence $qY/\ell \ge 1$ and $H \le 2qY/\ell$, and since $\rho^{\beta_2} D'
\le d_2 \cdot d_1 f d_2 \le X_2$ and $\ell \ge \theta N_1/(rD'\rho^2)$,
\[
  \frac{2HP_2}q + 1 \le \frac{4YP_2}\ell + 1 \le \frac{4Yr\rho^3 X_2}{\theta N_1} + 1.
\]
Moreover $H \le qYr\rho^2 D'/(\theta N_1) + 1 \le 8rq^{1/2 + y + s} + 1 < q/2$ for large $q$, since $D' \le (X_1
X_2)^{1/2} \le q^{L - 1/2}$ and $y + s \le 1/6$; and $HD \le 2qYr\rho^3 X_2/(\theta N_1) \le q^{L + 2}$, so $F$ bounds
the divisors that the lemma needs. Since $H + 1 > qY/\ell$, the second term of \eqref{eq:block-pair} has $(q/(2\ell(H +
1)))^r \le (2Y)^{-r} \le q^{-yr} \le q^{-2}$, whatever the size of $d_2$.
% Lean: SieveTypeIBSumWide.bdPW_le (the cut H = ceil(qY/l)), SieveTypeIBSumWide.witness_le_wide

\emph{The bound in powers of $q$.} On a block pair that holds a leaf, $P_1 P_2 n$ and $(B - A) P_1 P_2$ are at most
$\rho^2(1 + \theta) N_2 \le 16 q^L$, $P_1/q + 1 \le \rho X_1/q + 1 \le 3q^{L - 2 - \xi}$ since $\xi \le L - 2$, $n/q + 1 \le
(1 + \theta)\rho^2 N_2/(X_3 q) + 1 \le 17 q^{\delta'}$, and
\[
  F\Bigl(\frac{2HP_2}q + 1\Bigr) \le C_d q^s \bigl(64 r q^{y + s + c} + 1\bigr)
  \le (64r + 1) C_d \, q^{\xi - \delta' - \kappa + y + 2s},
\]
since $\max(c, 0) \le \xi - \delta' - \kappa$: this is where the lemma's condition on $c$ enters, and $c$ may be positive.
So the three terms of \eqref{eq:block-pair} are at most
\[
  \sqrt{816(64r + 1) C_d} \, q^{L - 1 - (\kappa - y - 2s)/2}, \qquad 16 q^{L - 2}, \qquad
  4C_d \, q^{L - 1/2 + s}/\varphi(q),
\]
the last since $P_1 P_2 \le \rho^2 (X_1 X_2)^{1/2} \le 4q^{L - 1/2}$ and $d(q) \le C_d q^s$; a block pair that holds no
leaf contributes nothing. The Fourier masses of at most $J_0$ cut-offs are at most $(J_0 + 1)(C q^s)^{J_0}$, with $C$
that of \S\ref{app:family}, and the fibres at most $q^{s/8}$ for large $q$. Since $\kappa - y - 2s \ge 3\kappa/4 - 2s
\ge (3J_0 + 16)s$ by the choice of $s$, the bilinear sums total at most $C'(\log q)^2 q^{L - 3s/4}/\varphi(q)$, with
$C'$ independent of $q$.
% Lean: SieveTypeIBAsymWide.bdUW_le_pow, SieveTypeIBSumWide.blocks_le_wide, SieveTypeIBSumWide.leaves_family_wide

\emph{The main terms.} The difference of the main terms is $(N_2 - N_1)/q \cdot \Gamma \cdot \sum I/d$, plus the
transition from the splines' masses, at most $(\Sigma_1 \Theta/q) \sum I/d + (4\Sigma_0/q + 2\Sigma_0 d(q)/\varphi(q))
\sum I$ with $\Theta \le 11 q^{L - s}$, $\Sigma_1 = \sum |\lambda|/f$ and $\Sigma_0 = \sum |\lambda|$. Since $(d_1
d_2)^2 \le X_1 X_2 \le q^{2L - 1}$ on the leaves, $\sum I \le q^{L - 1/2} \Sigma_*^{K_1 + K_2}$, where $\Sigma_* =
\log((L + 1)/\varepsilon_w) + 1$ bounds the sums of $1/p$ over the primes in $[w, N_2]$ for large $q$; and the counted
parts near the boundaries are at most $(12\theta + 1/w) \Sigma_*^{K_1 + K_2 - 1}$ times $2(1 + \theta)\rho^2 N_2
\Sigma_1/\varphi(q)$, by the pivots, since every prime is at least $w \ge q^s$. With $\Sigma_0 \le 2\,\mathrm{rem} \le
24 w^{2k + 7} \le q^{1/4}$, all these are at most $(\log q)^{C''} q^{L - 3s/4}/\varphi(q)$ for a constant $C''$, which
is below $q^{L - s/4}/\varphi(q)$ for large $q$, and so are the bilinear sums. With $|\Gamma| \le \zeta_k q/(\varphi(q)
\log w)$ this is the lemma, with $\delta = s/4$.
\end{proof}

\subsection*{Proof of Proposition~\ref{M-prop:harman-two}}
% Lean: SieveTwoComp.prop6_two_at (prop6_two); at one q SieveTwoP6.prop6_two_core, with SieveTwoP6.root_bound2,
% SieveTwoP6.stops_bound2, SieveTwoP6.tree_pattern2_ib, SieveTwoTerm.chainOK2_of_zone, SieveP6TreeLeaves.tree_leaves_le

\begin{proof}
\emph{The constants.} They are chosen before $q$ and depend only on the data of the proposition. Let $D_1$ and $D_2$ be
as in (d), and
\[
  c' = \tfrac1{32} \min\bigl(1,\ L - 1 - \max(D_1, 1),\ 2L - 1 - \max(D_1, 1) - D_1 - 2D_2\bigr),
\]
which is positive by (d) and $L > 2$. Put $M_0 = (\log((L + 1)/a_0) + 1)^j$. Since $(2k + 4)^2 \zeta_k \to 0$, there is
an even $k \ge 2$ with $2C_0^2 \le (k + 1)(k + 2)$, $c' < (2k + 4)a_0$ and $(2k + 4)^2 \zeta_k \le \eta c'^2/(4e(M_0 +
1)a_0)$. Put $\varepsilon_w = c'/(2k + 4)$, so that $0 < \varepsilon_w < a_0$, $(2k + 4)\varepsilon_w = c' \le 1/16$ and
$(2k + 7)\varepsilon_w \le 2c'$; and put $s_{\max} = \lceil (L + 2)/\varepsilon_w \rceil$, $\Lambda_1' = \Lambda_1 + L +
2$ and $\Lambda_0 = \Lambda_1' + (j + s_{\max})(L + 1)$. Let $\delta_1 > 0$ be the least of the $\delta$'s that
Lemma~\ref{M-lem:term-two} gives, with $e_0 = \varepsilon_w$, $J_0 + s_{\max} + 5$ comparisons and $\Lambda_1'$, for the
chains of the lengths up to $j + s_{\max} + 1$. Let $\delta_3 > 0$ be the least of those that Lemma~\ref{M-lem:leaves}
gives, with $\varepsilon_w$, $k$, $\Lambda_0$, $J_0 + s_{\max} + 5$ comparisons, the tuple lengths $K_1, K_2 \le j +
s_{\max}$, and the exponents $\xi = L - 1 - \max(D_1, 1) - 2c'$, $c = \xi - 20c'$, $\delta' = 4c'$ and $\kappa = 16c'$; they
satisfy its conditions, since $\xi \le L - 2$, $\xi - \delta' - \kappa = c = L - 1 - \max(D_1, 1) - 22c' \ge 0$ and $\delta'
= \kappa/4 \le 1/8$.

At $q$, put $w = \lceil q^{\varepsilon_w} \rceil$ and $z_0 = \lceil q^{a_0} \rceil$, so that $w \le z_0$, and $P_1 =
\lceil q^u \rceil$, $P_2 = \lfloor q^v \rfloor$, $R_- = \lceil q^{\rho_1} \rceil - 1$ and $R_+ = \lfloor q^{\rho_2}
\rfloor$. For large $q$:
\begin{enumerate}
\item[(i)] $(P_1 + 1) z_0 \le P_2$, since $(q^u + 2)(q^{a_0} + 1) \le 6q^{u + a_0} \le q^v$ by (a);
\item[(ii)] $q^{\sum_{\mathcal S_0} h_i} < P_2$, since $2q^{\sum_{\mathcal S_0} h_i} \le q^v$ by (a);
\item[(iii)] $P_2 R_+ \le N_1$, since $P_2 R_+ \le q^{v + \rho_2} \le q^L$ by (b);
\item[(iv)] $\lfloor N_1/R_+ \rfloor \ge 4q^{L - \rho_2 - \varepsilon}$ and $\lfloor N_2/(R_- + 1) \rfloor \le 2q^{L -
  \rho_1} \le q^{L - \rho_1 + \varepsilon}$, since $q^\varepsilon \ge 8$ and $q^{L - \rho_2} \ge 4$;
\item[(v)] $P_1 + 1 \le 4q^L \le q^{L + 2} \le w^{s_{\max}}$.
\end{enumerate}
By (iv), the region's hypothesis of Lemma~\ref{M-lem:term-two} with $A = P_1$, $B = P_2$ and these $R_-$, $R_+$ follows
from (c): an $m_1 \in [\log_q P_1, \log_q P_2]$ lies in $[u, v]$, and an $m_2$ with $\lfloor N_1/R_+ \rfloor \le 4q^{m_1
+ m_2}$ and $q^{m_1 + m_2} \le \lfloor N_2/(R_- + 1) \rfloor$ has $m_1 + m_2 \in [L - \rho_2 - \varepsilon, L - \rho_1 +
\varepsilon]$. By (b) and the boxes, every tuple of the family has $q^{\rho_1} \le \prod_{\mathcal S_r} r_i \le
q^{\rho_2}$, that is $R_- < \prod_{\mathcal S_r} r_i \le R_+$; and by (iii), $B R_+ \le N_1$.

\emph{Two patterns.} With $r_{\mathcal S_0} = \prod_{i \in \mathcal S_0} r_i$, split the tuples by $r_{\mathcal S_0} \le
P_1$ or $r_{\mathcal S_0} > P_1$: each part is a family of the same kind, with one more comparison, $\log P_1 -
\sum_{\mathcal S_0} \log r_i \ge 0$ or its strict negation.

\emph{$r_{\mathcal S_0} > P_1$.} Then $P_1 < r_{\mathcal S_0} \le q^{\sum_{\mathcal S_0} h_i} < P_2$ by (ii). By
Buchstab's identity, $S(\mathcal A_r, z_0) = S(\mathcal A_r, r_j) + \sum_{z_0 \le p < r_j} S(\mathcal A_{rp}, p)$, and
the same for $\mathcal B$. The first sum, over the tuples, and the second, over the $(j + 1)$-tuples $(r_1, \dots, r_j,
p)$ cut out by the comparisons and $z_0 \le p < r_j$, are of the kind of Lemma~\ref{M-lem:term-two}, sifted at the
chain's last prime, with $\mathcal S = \mathcal S_0$, the rest $\mathcal S_r$, $A = P_1$, $B = P_2$ and $R_-$, $R_+$:
the chains are strictly decreasing, with entries at least $z_0 \ge q^{\varepsilon_w}$, and the region holds. So this
pattern contributes at most $2q^{L - \delta_1}/\varphi(q)$.

\emph{$r_{\mathcal S_0} \le P_1$: the tree.} Apply \eqref{eq:tree} to each tuple, with $d_0 = r$, $z = z_0$, $\gamma =
r_{\mathcal S_0}$ and $E = P_1 + 1$: Buchstab's tree from $z_0$ down to $w$, stopped as soon as $r_{\mathcal S_0}$ times
the branch's primes exceeds $P_1$. A set $P$ of the tree with $r_{\mathcal S_0} \prod P < E$ has $|P| < s_{\max}$, since
$\prod P \ge w^{|P|}$ and $E \le w^{s_{\max}}$ by (v); so the leaves and the stops come in at most $s_{\max}$ sizes $s =
|P|$.

\emph{The stops.} A stop $(p, P)$ has $E \le r_{\mathcal S_0} \prod P \cdot p < E z_0 \le P_2$ by (i), since $p < z_0$:
its first variable $r_{\mathcal S_0} \prod P \cdot p$ lies in $(P_1, P_2]$. For each size $s$, the stops are the chains
$(r_1, \dots, r_j, y_1, \dots, y_s, p)$ of primes that satisfy the family's comparisons, the pattern's and
\[
  y_1 < z_0, \qquad y_1 > \dots > y_s > p \ge w, \qquad r_{\mathcal S_0} y_1 \cdots y_s < E \le r_{\mathcal S_0} y_1
  \cdots y_s \, p,
\]
at most $J_0 + s_{\max} + 5$ comparisons with constant terms at most $\Lambda_1' \log q$, sifted at their last prime
$p$; for $s = 0$ the condition $y_1 < z_0$ reads $p < z_0$. They are strictly decreasing, since $y_1 < z_0 \le r_j$,
their entries are at least $w \ge q^{\varepsilon_w}$, and with $\mathcal S = \mathcal S_0 \cup \{\text{the places of }
y_1, \dots, y_s, p\}$ and the rest $\mathcal S_r$ they have $A < \prod_{\mathcal S} \le B$ and $R_- < \prod_{\mathcal
S_r} \le R_+$. So Lemma~\ref{M-lem:term-two} bounds each size by $q^{L - \delta_1}/\varphi(q)$, and the stops total at
most $s_{\max} q^{L - \delta_1}/\varphi(q)$.

\emph{The leaves.} A leaf $P$ has $r_{\mathcal S_0} \prod P < P_1 + 1 \le 3q^u \le 3q^t q^{\sum_{\mathcal S_0} l_i} \le
3q^t r_{\mathcal S_0}$ by (a), so $\prod P < 4q^t$; its modulus is $d = r \prod P$, with the sign $(-1)^{|P|}$. Those
with $d \le q^{L - 1 - 4c'}$ are counted by \eqref{eq:leaf}, and total at most $(2\zeta_k (N_2 - N_1)/(\varphi(q) \log
w) + 24 w^{2k + 7} q^{L - 1 - 4c'}) \sum 1/d$. Of each size, the others are split into $d_1$, the product of the $r_i$
with $i \in \mathcal S_1$ and, if $\iota = 1$, of the primes of $P$, and $d_2$, the product of the rest: the parts of
the leaf's decreasing tuple on these places are decreasing tuples of primes in $[w, N_2]$, and $d_1 \le 4q^{D_1}$, $d_2
\le 4q^{D_2}$. Since $w^{2k + 4} \le 2^{2k + 4} q^{c'}$, for large $q$

\section{Proof of Lemma~\ref{M-lem:rough}}\label{app:rough}

\begin{proof}
Write $c_* = 0.565073$ and $\bar\omega = 0.5671433$, $u = \log 2y/\log z$, and say that an integer is $z$-rough if it has no
prime factor below $z$. Four facts are used.

\emph{(a) Prime sums.} By the prime number theorem with an error $o(t/\log t)$ (in the library, the bound
$\psi(t) - t = O(t^{11/12 + \varepsilon})$, which it derives from the case of $\zeta$ of Input~\ref{M-inp:half-plane};
the classical theorem would do), for every $\varepsilon_3 > 0$,
$|\theta(t) - t| \le \varepsilon_3 t/\log t$ for large $t$. With partial summation this gives, for large $a \le b$,
\[
  \sum_{a < p \le b} \frac1{p \log p} \le \frac1{\log a} - \frac1{\log b} + \frac{4\varepsilon_3}{\log^2 a},
\]
and, if $2 \log b \le \mathcal L$,
\[
  \sum_{a < p \le b} \frac1{p(\mathcal L - \log p)}
  \le \frac{1 + \varepsilon_3/\log a}{\mathcal L} \log \frac{\log b \, (\mathcal L - \log a)}{\log a \, (\mathcal L -
  \log b)} + \frac{6\varepsilon_3}{\mathcal L \log^2 a};
\]
and, for large $y$, there are at most $(1 + \varepsilon_3) y/\log 2y$ primes in $(y, 2y]$.

\emph{(b) Buchstab's inequality.} For $y \ge 2$,
\[
  \Phi(y, z) \le \#\{p : y < p \le 2y\} + \sum_{z \le p \le \sqrt{2y}} \Phi(y/p, p),
\]
since a $z$-rough integer $m \in (y, 2y]$ that is not prime has its least prime factor $p$ in $[z, \sqrt{2y}]$, and
$m/p$ is a $p$-rough integer in $(y/p, 2y/p]$; the map $m \mapsto (p, m/p)$ is injective. Also $\Phi(y, z) = 0$ if
$y \ge 1$ and $2y < z$, since every integer of $(y, 2y]$ is then at least $2$ and below $z$.

\emph{(c) Buchstab's function on $[2, 3]$.} For $u > 1$, $1 + \log(u - 1) \le \bar\omega u$: the inequality $\log t \le t - 1$ at
$t = \bar\omega(u - 1)$ gives $1 + \log(u - 1) \le \bar\omega(u - 1) - \log \bar\omega \le \bar\omega u$, since $-\log \bar\omega \le \bar\omega$, that is $\bar\omega e^{\bar\omega} \ge 1$, which
holds because $e^{\bar\omega} \ge \sum_{m \le 12} \bar\omega^m/m! > 1/\bar\omega$. So $\omega^* \le \bar\omega$ on $[2, 3]$, and since $c_* < \bar\omega$,
$\omega^* \le \bar\omega$ on $[2, \infty)$; and $\omega^* \le c_*$ on $[3, \infty)$, since $\omega^*(3) = (1 + \log 2)/3 < c_*$.

\emph{(d) A delay inequality.} Let $M(u) = \log(u - 1)$ for $2 \le u \le 3$ and
$M(u) = \log 2 + \bar\omega \min(u - 3, 1) + c_* \max(u - 4, 0)$ for $u > 3$, which is $\int_1^{u - 1} \omega^*(t)\, dt$ with
$\omega^*$ replaced by $\bar\omega$ on $[2, 3]$. Then $1 + M(u) \le u \omega^*(u)$ for $u \ge 2$. For $u \le 3$ it is an
equality. For $u > 3$, $\omega^*(u) = c_*$, and $1 + M(u) - c_* u$ is affine on $[3, 4]$, with slope $\bar\omega - c_* > 0$, and
constant on $[4, \infty)$; so it suffices that it is at most $0$ at $u = 4$, that is $1 + \log 2 + \bar\omega \le 4c_*$, which
holds since $\log 2 < 0.6931471808$ and $2.2602904808 < 2.260292 = 4c_*$.

We prove by induction on the integer $k \ge 0$ the statement $\mathrm C(k)$: for every $\varepsilon > 0$ there is
$z_1$ such that $\Phi(y, z) \le (\omega^*(u) + \varepsilon) y/\log z$ for all $z \ge z_1$ and all $y \ge 1$ with
$u \le k$. $\mathrm C(0)$ holds trivially, since $u > 0$, and the lemma is $\mathrm C(\lceil U \rceil)$. Assume
$\mathrm C(k)$, let $0 < \varepsilon \le 1$ (which suffices, since the bound for one $\varepsilon$ implies it for every
larger one) and $\varepsilon_3 = \varepsilon/100$, let $z$ be large in terms of
$\varepsilon$ and of the $z_1$ of $\mathrm C(k)$ for $\varepsilon/4$, and let $y \ge 1$ with
$\mathcal L = \log 2y = u \log z \le (k + 1) \log z$. If $2y < z$, then $\Phi(y, z) = 0$. Otherwise $u \ge 1$, $y \ge
z/2$ is large, and (a) and (b) give
\[
  \Phi(y, z) \le \Bigl(1 + \frac\varepsilon4\Bigr) \frac{y}{\mathcal L} + \sum_{z \le p \le \sqrt{2y}} \Phi(y/p, p).
\]
If $u < 2$, then $\sqrt{2y} < z$, the sum is empty, and
$\Phi(y, z) \le (1 + \varepsilon/4) y/(u \log z) \le (\omega^*(u) + \varepsilon) y/\log z$. Let $u \ge 2$. A prime
$p \in [z, \sqrt{2y}]$ has $y/p \ge 1$ and
$\log(2y/p) = \mathcal L - \log p \le k \log p$, since $\mathcal L \le (k + 1) \log p$. So $\mathrm C(k)$ gives
$\Phi(y/p, p) \le (\omega^*(u_p) + \varepsilon/4) \, y/(p \log p)$ with $u_p = (\mathcal L - \log p)/\log p$. The term
$p = z$, if $z$ is prime, contributes at most $2y/(z \log z)$, since $\omega^* \le 1$. By (a), the parts
$\varepsilon/4$ of the other terms contribute at most $(\varepsilon/4)(1/\log z + 4\varepsilon_3/\log^2 z) \, y$. For
the parts $\omega^*(u_p)$: if $p \le (2y)^{1/4}$ then $u_p \ge 3$ and $\omega^*(u_p) \le c_*$; if
$(2y)^{1/4} < p \le (2y)^{1/3}$ then $2 \le u_p < 3$ and $\omega^*(u_p) \le \bar\omega$; and if $(2y)^{1/3} < p \le (2y)^{1/2}$ then
$1 \le u_p < 2$ and $\omega^*(u_p)/(p \log p) = 1/(p(\mathcal L - \log p))$, by the definition of $\omega^*$ on
$[1, 2)$. By (a), applied on $(z, \sqrt{2y}]$ if
$u \le 3$, on $(z, (2y)^{1/3}]$ and $((2y)^{1/3}, \sqrt{2y}]$ if $3 < u \le 4$, and also with the first range split at
$(2y)^{1/4}$ if $u > 4$,
\[
  \sum_{z < p \le \sqrt{2y}} \frac{\omega^*(u_p)}{p \log p} \le \frac{M(u) + 12\varepsilon_3}{\mathcal L}
  + \frac{8\varepsilon_3}{\log z}.
\]
Collecting the terms, with $\mathcal L = u \log z$,
\[
  \Phi(y, z) \le \frac y{\log z} \Bigl(\frac{1 + M(u)}u + \frac{\varepsilon/4 + 12\varepsilon_3}u + \frac2z
  + 8\varepsilon_3 + \frac\varepsilon4 + \frac{\varepsilon \varepsilon_3}{\log z}\Bigr).
\]
By (d) the first term is at most $\omega^*(u)$, and since $u \ge 1$ and $z$ is large the others add up to at most
$0.96\varepsilon$. So $\Phi(y, z) \le (\omega^*(u) + \varepsilon) y/\log z$, which proves $\mathrm C(k + 1)$.
\end{proof}

\emph{A fact used in Section~\ref{M-sec:decomp}.} The bound $\Omega$ of that section, though not the proof above,
uses that on $[2, 3]$, $\omega^*(u) = (1 + \log(u - 1))/u$ increases up to $2.76$ and decreases from $2.77$ on. For
$2 \le u \le u' \le 2.76$, $\omega^*(u) \le \omega^*(u')$ is $(u' - u)(1 + \log(u - 1)) \le u \log((u' - 1)/(u - 1))$:
the right side is at least $u(u' - u)/(u' - 1)$, by $\log t \ge 1 - 1/t$, and the left at most $(u' - u)u/1.76$, by
$\log t \le t - 1$ at $t = (u - 1)/1.76$ and $\log 1.76 \le 1/1.76$. For $2.77 \le u \le u'$ the reverse inequality
follows in the same way from $\log((u' - 1)/(u - 1)) \le (u' - u)/(u - 1)$ and
$\log(u - 1) \ge \log 1.77 \ge 1/1.77 \ge 1/(u - 1)$. The two constants come from the exponential's series.

In Sections~\ref{M-sec:type1} to~\ref{M-sec:certificate}, \file{formalisations/harman-sieve.md} gives the sources of

\subsection*{The library's proof of Input~\ref{M-inp:half-plane}}

The library proves Input~\ref{M-inp:half-plane} as \lean{Eis.LFunction\_ne\_zero\_of\_re\_gt}, along the route of OpenAI's
companion preprint \cite{openai-companion}, on its own toolchain; it imports nothing from OpenAI's release
(\file{formalisations/openai-half-plane.md}). The route has five steps. Power savings for the smoothed Möbius sums of the
Dedekind zeta function of $\mathbb Q(\sqrt{-3})$ give the half-plane. The target is embedded in a family of sextic
twists, over which a mean square bounds it. Poisson summation and the Gauss--Jacobi identities turn the Möbius function
times sextic Gauss sums into cubic Gauss sums. Kubota's cubic theta function \cite{kubota} supplies the transformation,
with Patterson's formula for its coefficients \cite{patterson}, Dunn and Radziwi\l\l's cusp expansions
\cite{dunn-radziwill}, and the quadratic large sieve over $\mathbb Z[\omega]$ \cite{hb-quadratic, goldmakher-louvel},
which bounds the dual mean square. Möbius inversion removes the cube factors and gives the exponent $11/12$. The chain
gives $\zeta$ and $L(s, \chi_{-3})$ first, then the real characters, the characters of order dividing $6$, and every
other character through the family twisted by a character modulo $4q$. The classical results that the route uses,
among them cubic and sextic reciprocity \cite{ireland-rosen}, are theorems of the library. The import cone of the
theorem in the library has $223$ files, all of them in \texttt{lib/core}.

\subsection*{Independent checking of the input}

Independent checking of Input~\ref{M-inp:half-plane} is partial, and little of it is independent of OpenAI. OpenAI's
release has its own Lean formalisation of the main preprint's result. Its catalogue \texttt{lean/formalization.yaml}
\cite{openai-lean}, read on 25 October 2026, lists that preprint among the papers with a formalised main result, and
names the theorem \lean{OAI.DirichletCharacter.\allowbreak LFunction\_ne\_zero\_of\_seven\_eighths\_lt\_re}. Its
statement is for every modulus $q \ge 1$, every character mod $q$ and every $s$ with $\operatorname{Re} s > 7/8$,
against Mathlib's $L$-function, with the same exception at $s = 1$ for the trivial character, so it implies
Input~\ref{M-inp:half-plane} as stated here. It is OpenAI's own record, not an independent one, and the catalogue gives
its scope as partial progress and its review status as unchecked. The author read that release's statements,
scanned its Lean and rebuilt it on 7 October 2026, which showed its two main theorems to depend on Lean's standard
three axioms only, but did not read its proofs in full. This library's proof of Input~\ref{M-inp:half-plane} is a new
formalisation of the companion preprint's route, on a different Lean and Mathlib; it imports nothing from the release
and was not compared with it. It is a machine-checked proof of the companion preprint's theorem, but the route is
OpenAI's and the formalisation is the author's, written with the AI assistance described in the paper's acknowledgements, so
it is not a review of the mathematics that is independent of OpenAI and of the author. The check recorded by the tracker AIProofWatch \cite{aiproofwatch} on 25 October 2026 concerns the
specialisation to the Riemann zeta function at $7/8$, not Dirichlet $L$-functions in general. So the author knows of no
review of the mathematics for Dirichlet $L$-functions in general that is independent of OpenAI and of the author.

Huxley's estimate is refereed and classical. The library's proof of it is the author's, in the same way, and Lean's
kernel checks that proof; \file{formalisations/huxley-density.md} compares its statement with Huxley's.

\subsection*{What the kernel checks}

The library's own sources, the directories \texttt{lib/core}, \texttt{lib/pnt}, \texttt{lib/dh},
\texttt{lib/zeta23} and \texttt{lib/check}, contain no \lean{sorry}, no declared axiom and no \lean{native\_decide},
which \texttt{scripts/check\_hygiene.py} checks. The library vendors 19 files of PrimeNumberTheoremAnd \cite{pnt-plus}
and, in \texttt{lib/vendor/zeta23}, 117 files of the \texttt{zeta23} folder of Anthropic's \texttt{formal-math}
\cite{formal-math}, each ported to the
library's toolchain by a patch that the repository records. That scan does not cover them; but the import cone of
Theorem~\ref{M-thm:main} contains 15 of the PrimeNumberTheoremAnd files and none of the \texttt{zeta23} ones, and not
\texttt{Wiener.lean}, the vendored file that contains \lean{sorry}. In any case the axiom check of the theorems in the
table of Appendix~\ref{M-app:claims} covers the whole dependency cone of each, and a \lean{sorry} in it would appear there
as \lean{sorryAx}. The import cone of Theorem~\ref{M-thm:main} has $482$ files of the library, $282$ in
\texttt{lib/core} and $200$ in \texttt{lib/pnt}; it contains no module of the library's formalisation of Chen, Gupta and
Li's paper.

The certificate of Proposition~\ref{M-prop:certificate} is checked by kernel evaluation, \lean{decide +kernel}, and not
by compiled code. The search led by hints is evaluated on $1552$ parts, the theorems \lean{SieveAd.tree21500\_0} to
\lean{SieveAd.tree21500\_1551} in the $17$ files \file{lib/pnt/SieveHint21500P1.lean} to
\file{lib/pnt/SieveHint21500P17.lean}, each of which bounds the search's value on a range of cells; the $256$ steps of
Buchstab's identity that join the parts, and the root's test, are proved one by one in \file{lib/pnt/Linnik21500.lean},
each by \lean{decide +kernel}. The kernel checks each file in $187$ to $303$ seconds, about $76$ minutes in all; it
takes about $0.3$ ms on an evaluated box ($0.8$ ms over a frozen part), $1.2$ ms on a step of Buchstab's identity
($2.1$ ms), $0.25$ ms on a dropped box ($0.35$ ms) and $3$ ms on a reversal's root. The search reads every
configuration from the hints (\file{lib/pnt/SieveAdDirDefs.lean}), and its tests are written for the kernel: one pass
over a box by the recursor of lists, with the arithmetic of the natural numbers written out and the bits of the sets as
factors (\lean{SieveAd.fillTestsF}, \lean{SieveAd.frozTestsF}). They are proved equal to the tests of
Lemmas~\ref{M-lem:tests} and~\ref{M-lem:frozen-tests} (\lean{SieveAd.fillTestsF\_eq}, \lean{SieveAd.frozTestsF\_eq}),
so they are sound. The hints are $1552$ natural numbers, $5220964$ bytes in all, written in hexadecimal.
The definitions the kernel evaluates are arithmetic on Lean's core natural numbers, lists and booleans; their files
import from Mathlib only the notation $\mathbb N$. Lean's kernel reduces the operations of the natural numbers on
literals (addition, multiplication, division, comparison and the like) by the GMP library's arithmetic rather than by
their definitions, so this check trusts GMP as well as the kernel.

\subsection*{The artefact}

The library, the paper and this supplement are in the repository \cite{repo}; this version of the paper and of the
supplement describes the library at the commit that contains it. The earlier versions of the paper proved the exponent
$11/5$, by a Type~II estimate in one variable on a window with Chen, Gupta and Li's density estimate as a second input;
that theorem, \lean{Linnik22000E.linnik\_22000e}, is still in the library, and that version of the paper is in the
repository's history at commit \texttt{49dc981}. Before it, the paper proved $278/125$
(\lean{Linnik22240R.linnik\_22240r}, at commit \texttt{8ba053b}). The toolchain is Lean \texttt{v4.35.0-rc3}
\cite{lean4}, a release candidate, with Mathlib \cite{mathlib} at commit \texttt{0f64d30a},
PrimeNumberTheoremAnd at \texttt{650d312} and \texttt{formal-math} at \texttt{fbdc36b}. The repository's README gives the

from source, or replaying the environment with \texttt{lean4checker}, removes that trust), \texttt{scripts/build.sh} for
the build, which limits how many of the memory-hungry kernel checks of the certificate run at once, and

change to the library.

\section{Computations around the exponent}\label{sup:computations}

This section gives the computations behind the remarks of Section~\ref{M-sec:region} on the region, and behind the last
paragraph of Section~\ref{M-sec:comparison} of the paper. They are computations with the definitions of the region and
of the certificate, by the scripts of \file{tools/linnik/}, and not theorems of the library; nothing in the paper or the
library depends on them.

\subsection*{The region}

\texttt{gen\_zone.py --exact} computes, for one variable shorter than $q$ and the other of length at least $q$, the
largest $s_1$ for which the zone condition of Definition~\ref{M-def:zone} holds for $g_{\mathrm H}$ at each of $41$
points of every piece, and compares it with the region's $\lambda\rho - \mu$: at $\rho = 0.01$, $0.05$, $0.1$, $0.2$ and
$0.3$ the two agree to six decimals. With the Pólya--Vinogradov bound $|k_Y(\psi)| \ll q^{1/2 + \varepsilon}$ in place of
the kernel bound, the Hal\'asz--Montgomery case reads $\Delta_i = \max(0, g(\beta) - (1/2 - \varepsilon) + s_i)$, and
the $136$ cases of the proof of Proposition~\ref{M-prop:region} hold at the vertices of the region exactly for
$\varepsilon \le 1/2 - g_{\mathrm H}(\beta^\dagger) = 0.0037454\ldots$ (\texttt{gen\_zone.py --pv}): beyond, the curve's
value at the threshold exceeds $1/2 - \varepsilon$. The threshold $7789/10000$ is the one for which the certificate's
tests are written, with $\lambda = 2(1 - \beta^\dagger - \varpi)$. Huxley's exponent itself brings the curve below
$T_* = 249/500$ from $\beta = 3.498/4.494 = 0.77837\ldots$ on, and a first step from $7784/10000$, with $A = 2.247$ and
$A(1 - \beta) = 0.497935 < 249/500$, would allow $\lambda = 0.443192$, a region about $0.2\%$ wider; a step from
$7783/10000$ has the value $0.49827$ there, too large. The paper does not use this.

\subsection*{The search and its plan}

The plan of Proposition~\ref{M-prop:certificate} is made by \file{tools/linnik/hint\_cert.py}, which runs the search of
Section~\ref{M-sec:certificate} in Python, definition by definition and in exact integers, with the strategy of that
section, and keeps the nodes that it takes. At $43/20$ it visits $34356403$ boxes, $28257594$ of them over the frozen
parts of reversals, in about nine minutes on four cores; its plan has $3400745$ steps, the nodes of the trees over the
frozen parts included, of which $161674$ are reversals through the largest prime. Before it writes a part, the script runs the part through its own copy of the search led by hints, which the
kernel then evaluates. The plan joins consecutive cells that one test evaluates, and consecutive dropped cells while the
joined cell's cost stays within $0.2\%$ of the sum of theirs, so the certificate's cells are unions of the strategy's.
These numbers are the script's, written into the header of \file{lib/pnt/Linnik21500.lean}; the cost $c$ and the
number of parts are the Lean's.

\subsection*{Below $43/20$}

The plan of the same search passes at $L = 2.145$ with cells of side $0.0035$: its cost is $0.975\mathcal K$, on
$6.0 \cdot 10^7$ boxes, with $6.6 \cdot 10^6$ steps, which would be about ten MB of hints and two and a half hours of
the kernel's time; that plan is not in the library. With each test of the search taken at the midpoint of its cell in place of the whole cell,
which is no certificate, the cost at $43/20$ on twenty of the first prime's cells falls by $24.6\%$; with those of one kind alone, by $12.1\%$ (the estimate
in two variables over the frozen parts of the reversals), $5.8\%$ (the same in the main tree), $5.8\%$ (the comparisons at
a floor in the main tree) and $4.1\%$ (those over the frozen parts). So about a quarter of the cost is the cells'. Each $0.005$ of the exponent costs between $5\%$ and
$9\%$: on cells of side $0.005$ the cost is $0.829\mathcal K$, $0.981\mathcal K$ and $1.034\mathcal K$ at $2.165$, $2.155$
and $2.15$. The rules that the certificate lacks are worth a few per cent each, measured in exact integers at $43/20$
on twenty of the first prime's cells, where the plan costs $0.15403\mathcal K$ (\file{tools/linnik/hint\_probe.py}): a
second level of reversal, a box of the sign $-$ in the tree over a reversed box's frozen part reversed again through
the largest frozen prime, saves $2.3\%$ for $2.6$ times the search; the reversal of boxes of the sign $+$, which the
library proves (\lean{SieveAd.bd\_flip\_pos}) but the search does not use, $2.3\%$ for twice the search; and a plan that
weighs each step's cost against its time in the kernel, about $5\%$ at equal time. By these numbers, which are
estimates (no plan was made at $2.14$), $2.14$ needs the last two and cells of side about $0.003$ together, on a
certificate two to three times this one; nothing measured gives $2.13$, and $2.12$ is beyond these rules. In
particular a further reversal of the boxes of two primes that the plan drops, which are $42\%$ of its cost, does not
pay: on nine cells of the first prime from $x^{0.16}$ to $x^{0.47}$, none of them costs less reversed than dropped,
with one level of reversal or two.
